  1588→  /** Зарегистрировать предмет, если его ещё нет (паттерн fire.js). */
  1589→  function addIfMissing(items, o) {
  1590→    if (!o || !o.id || items[o.id]) return;
  1591→    items[o.id] = o;
  1592→  }
