   405→  /** Снег: n div'ов с рандомными позицией, размером, скоростью. */
   406→  function makeSnow(parent, n) {
   407→    for (var i = 0; i < n; i++) {
   408→      var f = el('div', 'menu-flake', parent);
   409→      var size = 2 + Math.random() * 5;                     // px
   410→      var dur = 7 + Math.random() * 11;                     // сек падения
   411→      var delay = -Math.random() * 18;                       // старт «из середины»
   412→      var sway = 10 + Math.random() * 26;                   // размах сноса
   413→      f.style.left = (Math.random() * 100).toFixed(2) + '%';
   414→      f.style.width = size.toFixed(1) + 'px';
   415→      f.style.height = size.toFixed(1) + 'px';
   416→      f.style.opacity = (0.25 + Math.random() * 0.6).toFixed(2);
   417→      f.style.animationDuration = dur.toFixed(2) + 's';
   418→      f.style.animationDelay = delay.toFixed(2) + 's';
   419→      f.style.setProperty('--sway', sway.toFixed(1) + 'px');
   420→    }
   421→  }

   426→    card.setAttribute('data-id', prof.id);
   427→    card.setAttribute('data-num', num || '·');
   428→    card.title = 'клавиша ' + (num || '·');
   429→
   430→    var nm = el('div', 'menu-card-name', card);
   431→    setText(nm, prof.name);
   432→
   433→    var ds = el('div', 'menu-card-desc', card);
   434→    setText(ds, prof.desc);
   435→
   436→    // груз: компактной строкой; неизвестные предметы — тихо пропускаем
   437→    // (датафайлы идут параллельными агентами, id могут ещё ехать)
   438→    var known = [];
   439→    var unknown = [];
   440→    if (prof.items && prof.items.length) {
   441→      for (var i = 0; i < prof.items.length; i++) {
   442→        var it = prof.items[i];
   443→        if (!itemKnown(it.id)) { unknown.push(it.id); continue; }
   444→        known.push(itemName(it.id) + (it.n > 1 ? ' ×' + it.n : ''));
   445→      }
   446→    }
   447→    if (known.length) {
   448→      var inv = el('div', 'menu-card-items', card);
   449→      setText(inv, known.join(', '));
   450→    } else if (unknown.length) {
   451→      var inv0 = el('div', 'menu-card-items none', card);
   452→      setText(inv0, 'с пустыми руками');
   453→    } else {
   454→      var inv2 = el('div', 'menu-card-items none', card);
   455→      setText(inv2, 'ничего, кроме крепких кулаков');
   456→    }
   457→    if (unknown.length) {
   458→      WS.warn('menu: профессия «' + prof.id + '» ждёт предметы: ' + unknown.join(', '));
   459→    }
   460→
   461→    // навыки: чипы
   462→    var sk = prof.skills || {};
   463→    var keys = [];
   464→    for (var k in sk) if (sk.hasOwnProperty(k)) keys.push(k);
   465→    if (keys.length) {
   466→      var chips = el('div', 'menu-card-skills', card);
   467→      for (var j = 0; j < keys.length; j++) {
   468→        var chip = el('span', 'menu-chip', chips);
   469→        setText(chip, skillName(keys[j]) + ' ' + sk[keys[j]]);
   470→      }
   471→    }
   472→
   473→    card.addEventListener('click', function () {
   474→      snd('select');
   475→      profSel = (profSel === prof.id) ? null : prof.id;
   476→      refreshCards();
   477→      refreshStartState();
   478→    });
   479→    cardEls[prof.id] = card;
   480→  }

   506→  /** Переключить черту с правилами баланса. */
   507→  function toggleTrait(tr) {
   508→    if (tr.neg) {
   509→      if (negSel[tr.id]) delete negSel[tr.id];
   510→      else negSel[tr.id] = true;
   511→    } else {
   512→      if (posSel[tr.id]) {
   513→        delete posSel[tr.id];
   514→      } else {
   515→        var n = countKeys(posSel);
   516→        if (n >= MAX_POS) {
   517→          // лимит: тихий отказ, строка подсказки
   518→          setText(els.traitHint, 'сильных сторон не больше ' + MAX_POS + ' — снимите одну');
   519→          return;
   520→        }
   521→        posSel[tr.id] = true;
   522→      }
   523→    }
   524→    refreshTraits();
   525→    refreshStartState();
   526→  }

   832→  /** «Начать»: собрать выбор и создать партию. */
   833→  function onStart() {
   834→    if (!built) return;
   835→    var nn = countKeys(negSel);
   836→    if (!profSel) { showErr('Выберите профессию.'); return; }
   837→    if (nn !== NEED_NEG) {
   838→      showErr('Слабостей должно быть ровно ' + NEED_NEG + ': зима не спрашивает согласия.');
   839→      return;
   840→    }
   841→
   842→    var prof = null;
   843→    for (var i = 0; i < PROFESSIONS.length; i++) {
   844→      if (PROFESSIONS[i].id === profSel) prof = PROFESSIONS[i];
   845→    }
   846→    if (!prof) { showErr('Профессия не найдена.'); return; }
   847→
   848→    var traits = [];
   849→    for (var k = 0; k < TRAITS.length; k++) {
   850→      var t = TRAITS[k];
   851→      if (t.neg ? negSel[t.id] : posSel[t.id]) traits.push(t.id);
   852→    }
   853→
   854→    snd('bell');
   855→    var okNew = false;
   856→    try { okNew = WS.Main.newGame(prof.id, traits, parseSeed()); } catch (e) {
   857→      WS.err('menu newGame', e && e.message);
   858→      okNew = false;
   859→    }
   860→    if (!okNew) { showErr('Не удалось начать игру.'); return; }
   861→
   862→    applyProfession(prof);
   863→    hide();
   864→  }

   883→  function applyProfession(prof) {
   884→    var g = WS.G;
   885→    if (!g || !g.player) return;
   886→    var p = g.player;
   887→
   888→    // навыки: поднимаем до профилированного уровня (не выше 10)
   889→    var sk = prof.skills || {};
   890→    for (var k in sk) {
   891→      if (!sk.hasOwnProperty(k)) continue;
   892→      if (p.skills && p.skills[k]) {
   893→        p.skills[k].lv = Math.max(p.skills[k].lv || 0, Math.min(10, sk[k]));
   894→      }
   895→    }
   896→
   897→    // груз: через инвентарь (вес/стаки/события); неизвестные — мимо
   898→    var items = prof.items || [];
   899→    for (var i = 0; i < items.length; i++) {
   900→      var it = items[i];
   901→      if (!itemKnown(it.id)) continue;
   902→      var given = 0;
   903→      try {
   904→        if (WS.Inv && WS.Inv.give) given = WS.Inv.give(it.id, it.n || 1);
   905→      } catch (e2) { given = 0; }
   906→      if (!given) {
   907→        try {
   908→          if (WS.Action && WS.Action.give) WS.Action.give(it.id, it.n || 1);
   909→        } catch (e3) { /* совсем без инвентаря — молчим */ }
   910→      }
   911→    }
   912→  }

   914→  /** Разбор зерна мира: '' -> undefined; число; иначе хэш строки. */
   915→  function parseSeed() {
   916→    if (!els.seed) return undefined;
   917→    var v = String(els.seed.value || '').trim();
   918→    if (!v) return undefined;
   919→    if (/^\d+$/.test(v)) return (parseInt(v, 10) >>> 0);
   920→    // строка — детерминированный хэш (одна и та же — один и тот же мир)
   921→    try {
   922→      if (WS.RNG && WS.RNG.strHash) return (WS.RNG.strHash(v) >>> 0);
   923→    } catch (e) { /* ниже */ }
   924→    var h = 2166136261;
   925→    for (var i = 0; i < v.length; i++) {
   926→      h ^= v.charCodeAt(i);
   927→      h = (h * 16777619) >>> 0;
   928→    }
   929→    return h;
   930→  }

   932→  /** «Случайный герой»: судьба решает за игрока. */
   933→  function onRandomHero() {
   934→    if (!built) return;
   935→    snd('select');
   936→    // профессия
   937→    var p = PROFESSIONS[Math.floor(Math.random() * PROFESSIONS.length)];
   938→    profSel = p.id;
   939→    // сильные черты: 0..MAX_POS случайных
   940→    posSel = {};
   941→    var pos = [];
   942→    var negs = [];
   943→    var i;
   944→    for (i = 0; i < TRAITS.length; i++) {
   945→      if (TRAITS[i].neg) negs.push(TRAITS[i]); else pos.push(TRAITS[i]);
   946→    }
   947→    var wantPos = Math.floor(Math.random() * (MAX_POS + 1));
   948→    shuffle(pos);
   949→    for (i = 0; i < wantPos; i++) posSel[pos[i].id] = true;
   950→    // слабости: ровно NEED_NEG
   951→    negSel = {};
   952→    shuffle(negs);
   953→    for (i = 0; i < NEED_NEG; i++) negSel[negs[i].id] = true;
   954→
   955→    refreshCards();
   956→    refreshTraits();
   957→    refreshStartState();
   958→  }

   960→  /** Тасование Фишера—Йетса. */
   961→  function shuffle(arr) {
   962→    for (var i = arr.length - 1; i > 0; i--) {
   963→      var j = Math.floor(Math.random() * (i + 1));
   964→      var t = arr[i]; arr[i] = arr[j]; arr[j] = t;
   965→    }
   966→    return arr;
   967→  }

   969→  /** Клавиатура меню: стрелки, 1—6, Enter, R. */
   970→  function onMenuKey(ev) {
   971→    var code = ev.keyCode;
   972→    var i;
   973→    if (code === 37 || code === 39) {           // влево/вправо — профессии
   974→      ev.preventDefault();
   975→      var idx = -1;
   976→      for (i = 0; i < PROFESSIONS.length; i++) {
   977→        if (PROFESSIONS[i].id === profSel) idx = i;
   978→      }
   979→      var dir = (code === 39) ? 1 : -1;
   980→      idx = (idx + dir + PROFESSIONS.length) % PROFESSIONS.length;
   981→      profSel = PROFESSIONS[idx].id;
   982→      snd('select');
   983→      refreshCards();
   984→      refreshStartState();
   985→      return;
   986→    }
   987→    if (code >= 49 && code <= 54) {              // 1..6 — прямой выбор
   988→      i = code - 49;
   989→      if (PROFESSIONS[i]) {
   990→        profSel = PROFESSIONS[i].id;
   991→        snd('select');
   992→        refreshCards();
   993→        refreshStartState();
   994→      }
   995→      return;
   996→    }
   997→    if (code === 13) {                          // Enter — начать
   998→      if (els.btnStart && !els.btnStart.disabled) onStart();
   999→      return;
  1000→    }
  1001→    if (code === 82 || code === 114) {          // R — случайный герой
  1002→      onRandomHero();
  1003→    }
  1004→  }
