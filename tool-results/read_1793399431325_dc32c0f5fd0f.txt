   200→  /** Нож (тег cutting/skinning) в инвентаре или руке. */
   201→  function hasKnife() {
   202→    var p = P();
   203→    if (!p) return null;
   204→    var ids = [];
   205→    try {
   206→      if (p.inv) {
   207→        for (var i = 0; i < p.inv.length; i++) {
   208→          var it = p.inv[i];
   209→          if (it && it.id && ids.indexOf(it.id) < 0) ids.push(it.id);
   210→        }
   211→      }
   212→    } catch (e) { /* пусто */ }
   213→    var eq = p.equip;
   214→    if (eq) {
   215→      if (eq.hand1) ids.push(eq.hand1.id);
   216→      if (eq.hand2) ids.push(eq.hand2.id);
   217→    }
   218→    for (var k = 0; k < ids.length; k++) {
   219→      var def = (WS.Data && WS.Data.items) ? WS.Data.items[ids[k]] : null;
   220→      if (def && def.tags) {
   221→        if (def.tags.indexOf('cutting') >= 0 || def.tags.indexOf('skinning') >= 0) return ids[k];
   222→      }
   223→    }
   224→    return null;
   225→  }

   228→  function toScreenPos(wx, wy) {
   229→    var cam = WS.Cam;
   230→    if (!cam) return { x: 0, y: 0 };
   231→    var s = WS.M.toScreen(wx - cam.cx - (cam._shakeX || 0), wy - cam.cy - (cam._shakeY || 0));

   237→  var footAng = null;
   238→  function angForDir(dir) {
   239→    if (!footAng) {
   240→      footAng = [];
   241→      for (var d = 0; d < 8; d++) {
   242→        var s = WS.M.toScreen(DIRV[d][0], DIRV[d][1]);

   249→  /** Полоска прогресса над точкой (рисуется в мировых координатах). */
   250→  function drawBar(ctx, x, y, zoom, label, pct, col) {
   251→    try {
   252→      var w = 36 * zoom, h = 5 * zoom;
   253→      pct = clamp(pct, 0, 1);
   254→      ctx.fillStyle = 'rgba(8,12,22,0.78)';
   255→      ctx.fillRect(x - w / 2 - zoom, y - zoom, w + 2 * zoom, h + 2 * zoom);
   256→      ctx.fillStyle = col || '#d8a45c';
   257→      ctx.fillRect(x - w / 2, y, w * pct, h);
   258→      if (label) {
   259→        ctx.font = 'bold ' + Math.max(8, Math.round(9 * zoom)) + 'px sans-serif';
   260→        ctx.textAlign = 'center';
   261→        ctx.fillStyle = '#e8e2d0';
   262→        ctx.fillText(label, x, y - 3 * zoom);
   263→        ctx.textAlign = 'left';
   264→      }
   265→    } catch (e) { /* рисование — косметика */ }
   266→  }

   471→  function placeTrack(e, x, y, dx, dy) {
   472→    var kind = e.kind;
   473→    var spr = TRACK_SPR[kind];
   474→    if (!spr) return;
   475→    var dir = WS.M.dir8(dx, dy);
   476→    var rec = {

   479→    };

   481→      x: x, y: y, sprName: null, solid: false, shadow: false,

   483→      kind: kind, dir: dir, tmin: rec.tmin, fspr: spr, _rec: rec
   484→    });
   485→    o.render = trackRender;

   488→    if (trks.length > TRACK_CAP) {

   491→    }
   492→    // чутьё следопыта: ур.7 — подсказка игроку о свежем следе рядом
   493→    try {
   494→      var p = P();
   495→      if (p && trackLv() >= LV_TOAST &&
   496→          WS.M.dist(p.x, p.y, x, y) < 10 && kind !== 'bandit') {
   497→        toastOnce('trackhint' + kind,
   498→          t9('hunt.track_leads', 'След ведёт на ' + DIR_NAME[dir] + '…'), 'info', 9000);
   499→      }
   500→    } catch (err) { /* пусто */ }
   501→  }

   503→  function trackRender(ctx, sx, sy, zoom) {
   504→    try {
   505→      var o = this;
   506→      var img = WS.Spr.get(o.fspr, 0, o.dir);
   507→      if (!img) return;
   508→      var ageMin = WS.Time.totalMin() - o.tmin;
   509→      var fr = clamp(1 - ageMin / TRACK_FRESH_MIN, 0, 1); // 1 — свежий
   510→      ctx.globalAlpha = 0.35 + 0.55 * fr;
   511→      ctx.drawImage(img,
   512→        sx - img.width * zoom / 2,
   513→        sy - img.height * zoom / 2,
   514→        img.width * zoom, img.height * zoom);
   515→      ctx.globalAlpha = 1;
   516→      // ур.5: стрелка направления движения
   517→      if (trackLv() >= LV_SEE_DIR) {
   518→        ctx.save();
   519→        ctx.translate(sx, sy);
   520→        ctx.rotate(angForDir(o.dir));
   521→        ctx.strokeStyle = 'rgba(120,180,220,0.85)';
   522→        ctx.lineWidth = Math.max(1, 1.4 * zoom);
   523→        ctx.beginPath();
   524→        ctx.moveTo(0, 6 * zoom);
   525→        ctx.lineTo(0, -6 * zoom);
   526→        ctx.moveTo(-2.6 * zoom, -3.2 * zoom);
   527→        ctx.lineTo(0, -6 * zoom);
   528→        ctx.lineTo(2.6 * zoom, -3.2 * zoom);
   529→        ctx.stroke();
   530→        ctx.restore();
   531→      }
   532→    } catch (e) { /* след — косметика */ }
   533→  }

   551→  /** Тик следов: замер пути существ + старение + видимость. */
   552→  function tracksTick(dt) {
   553→    if (!running()) return;
   554→    var w = WS.World;
   555→    var list = w.ents;
   556→    var lv = trackLv();
   557→    var i, e;
   558→
   559→    // --- отпечатки: каждые TRACK_STEP тайлов пути ---
   560→    for (i = 0; i < list.length; i++) {
   561→      e = list[i];
   562→      if (!e || e.dead || !TRACK_KINDS[e.kind]) continue;
   563→      if (e.x === undefined || e.y === undefined) continue;
   564→      if (e._hx === undefined) {
   565→        e._hx = e.x; e._hy = e.y; e._path = 0;
   566→        continue;
   567→      }
   568→      var dx = e.x - e._hx, dy = e.y - e._hy;
   569→      var d = Math.sqrt(dx * dx + dy * dy);
   570→      if (d < 0.03) continue;
   571→      e._path = (e._path || 0) + d;
   572→      if (e._path >= TRACK_STEP) {
   573→        e._path = 0;
   574→        placeTrack(e, e._hx + dx * 0.5, e._hy + dy * 0.5, dx, dy);
   575→      }
   576→      e._hx = e.x; e._hy = e.y;
   577→    }
   578→
   579→    // --- старение и видимость ---
   580→    var now = WS.Time.totalMin();
   581→    for (i = trks.length - 1; i >= 0; i--) {
   582→      var rec = trks[i];
   583→      if (!rec.o || rec.o.removed) { trks.splice(i, 1); continue; }
   584→      var age = now - rec.tmin;
   585→      if (age >= TRACK_FRESH_MIN) { removeTrack(rec); continue; }
   586→      rec.o.hidden = !canSeeTrack(lv, age);
   587→    }
   588→
   589→    // --- кровавый след раненых зверей ---
   590→    bloodTick(dt);
   591→
   592→    // --- активный зов манка ---
   593→    callTick(dt);
   594→
   595→    // --- длительная работа (разделка) ---
   596→    actionTick(dt);
   597→  }

   599→  /** Информация о следе на тайле (для E-осмотра и чужих модулей). */
   600→  function trackInfo(x, y) {
   601→    var best = null, bd = 0.9;
   602→    var now = WS.Time.totalMin();
   603→    for (var i = 0; i < trks.length; i++) {
   604→      var rec = trks[i];
   605→      if (!rec.o || rec.o.removed) continue;
   606→      var d = WS.M.dist(rec.x, rec.y, x, y);
   607→      if (d < bd) { bd = d; best = rec; }
   608→    }
   609→    if (!best) return null;
   610→    var age = now - best.tmin;
   611→    var fresh = age <= 10 ? 'Свежий' : (age <= TRACK_FRESH_SOON ? 'Недавний' : 'Старый');
   612→    return {
   613→      kind: best.kind,
   614→      dir: best.dir,
   615→      dirName: DIR_NAME[best.dir],
   616→      ageMin: age,
   617→      fresh: fresh,
   618→      o: best.o
   619→    };
   620→  }

   649→  function isHuntTarget(o) {

   654→  /** Мой объект под курсором (в пределах досягаемости) или рядом с игроком. */
   655→  function findMyTarget() {
   656→    var w = WS.World;
   657→    var p = P();
   658→    if (!w || !p) return null;
   659→    var m = WS.Input.mouse();
   660→    var i, k, list, o, d;
   661→
   662→    // 1) под курсором
   663→    if (m && m.tx >= 0 && m.ty >= 0 && w.inB(m.tx, m.ty)) {
   664→      list = w.objsAt(m.tx, m.ty);
   665→      for (i = 0; list && i < list.length; i++) {
   666→        o = list[i];
   667→        if (isHuntTarget(o) && WS.M.dist(p.x, p.y, o.x, o.y) <= INTERACT_R) return o;
   668→      }
   669→    }
   670→    // 2) ближайший вокруг игрока
   671→    var ptx = Math.floor(p.x), pty = Math.floor(p.y);
   672→    var best = null, bd = INTERACT_R;
   673→    for (var cy = pty - 2; cy <= pty + 2; cy++) {
   674→      for (var cx = ptx - 2; cx <= ptx + 2; cx++) {
   675→        list = w.objsAt(cx, cy);
   676→        if (!list) continue;
   677→        for (k = 0; k < list.length; k++) {
   678→          o = list[k];
   679→          if (!isHuntTarget(o)) continue;
   680→          d = WS.M.dist(p.x, p.y, o.x, o.y);
   681→          if (d <= bd) { bd = d; best = o; }
   682→        }
   683→      }
   684→    }
   685→    return best;
   686→  }

   741→  /** E по трупу: сперва обыск (Loot.search), потом разделка. */
   742→  function corpseInteract(o) {

   765→    // старая туша: мясо в контейнере превращаем в тухлое
   766→    var spoiled = applySpoil(o);

   769→      try { WS.Ev.emit('inv:changed', {}); } catch (e) { /* пусто */ }
   770→    }
   771→    // подсказка про разделку, если есть нож и что забирать
   772→    if (hasKnife() && o.inv && o.inv.length) {
   773→      toastOnce('butcherhint' + o.id,
   774→        t9('hunt.butcher_hint', 'E у туши — разделать ножом (забрать всё)'), 'info', 6000);

   776→  }

   778→  /** Порча: мясо в контейнере старой туши становится meat_spoiled. */
   779→  function applySpoil(o) {
   780→    if (!o || !o.corpse) return 0;
   781→    var rec = corpseRec(o);
   782→    if (!rec) return 0;
   783→    var ageDays = (WS.Time.totalMin() - rec.deathMin) / 1440;
   784→    if (ageDays <= SPOIL_DAYS) return 0;
   785→    if (!o.inv || !o.inv.length) return 0;
   786→    var n = 0;
   787→    for (var i = 0; i < o.inv.length; i++) {
   788→      var it = o.inv[i];
   789→      if (it && it.id === 'meat_raw') {
   790→        it.id = 'meat_spoiled';
   791→        n += (it.count || 1);
   792→      }
   793→    }
   794→    return n;
   795→  }

   797→  /** Разделка: длительное действие с полоской. */
   798→  function startButcher(o) {
   799→    if (action) {
   800→      toastOnce('busy', t9('hunt.busy', 'Руки заняты'), 'warn');
   801→      return;
   802→    }
   803→    var knife = hasKnife();
   804→    if (!knife) {
   805→      toast(t9('hunt.need_knife', 'Без ножа тушу не разделать'), 'warn');
   806→      sndAt('error', o.x, o.y, 0.5);
   807→      return;
   808→    }
   809→    var kind = kindFromCorpse(o);
   810→    var size = (kind && KIND_SIZE[kind]) || 1;
   811→    var dur = BUTCHER_DUR[Math.max(1, Math.min(4, size))] || 3;
   812→    var mk = makeMarker(o.x, o.y, function () {
   813→      return t9('hunt.butchering', 'Разделка…');
   814→    }, function () {
   815→      return action && action.o === o ? action.t / action.dur : 0;
   816→    }, function () { return '#b0533a'; });
   817→    action = { type: 'butcher', o: o, kind: kind, size: size, t: 0, dur: dur, mk: mk };
   818→    sndAt('hit_flesh', o.x, o.y, 0.7);
   819→    fxText(t9('hunt.butchering', 'Разделка…'), o.x, o.y - 1, { col: '#c99a86', size: 13 });
   820→  }

   822→  /** Финиш разделки: забрать всё, XP, заменить костьми. */
   823→  function finishButcher() {
   824→    var a = action;
   825→    action = null;
   826→    dropMarker(a.mk);
   827→    var o = a.o;
   828→    if (!o || o.removed) return;
   829→
   830→    // лениво наполнить контейнер туши (если ещё не обыскана)
   831→    try {
   832→      if (o.container && WS.Loot && WS.Loot.fill) WS.Loot.fill(o);
   833→    } catch (e) { /* пусто */ }
   834→    if (!o.inv) o.inv = [];
   835→    if (!o.inv.length && !o.container) {
   836→      // тушки без таблицы лута: ворона, рыба
   837→      if (a.kind === 'crow') o.inv.push({ id: 'feather_bunch', count: 1 });
   838→      else if (a.kind === 'fish') o.inv.push({ id: 'fresh_fish', count: 1 });
   839→    }
   840→    applySpoil(o);
   841→
   842→    // забрать всё (что не влезло — на снег)
   843→    var taken = 0, dropped = 0;
   844→    for (var i = o.inv.length - 1; i >= 0; i--) {
   845→      var it = o.inv[i];
   846→      if (!it) continue;
   847→      var given = giveItem(it.id, it.count || 1);
   848→      taken += given;
   849→      if (given < (it.count || 1)) {
   850→        try {
   851→          if (WS.Loot && WS.Loot.dropItem) {
   852→            WS.Loot.dropItem(it.id, (it.count || 1) - given, o.x, o.y);
   853→            dropped++;
   854→          }
   855→        } catch (e2) { /* пусто */ }
   856→      }
   857→      o.inv.splice(i, 1);
   858→    }
   859→    // кости — всегда небольшой довесок

   973→  function bloodTick(dt) {
   974→    var w = WS.World;
   975→    var now = WS.Time.totalMin();
   976→    var list = w.ents;
   977→    for (var i = 0; i < list.length; i++) {
   978→      var e = list[i];
   979→      if (!e || e.dead) continue;
   980→      if (e._bleedUntil === undefined || e._bleedUntil < now) continue;
   981→      if (e.x === undefined || e._bx === undefined) { e._bx = e.x; e._by = e.y; continue; }
   982→      var dx = e.x - e._bx, dy = e.y - e._by;
   983→      var d = Math.sqrt(dx * dx + dy * dy);
   984→      if (d < 0.03) continue;
   985→      e._bpath = (e._bpath || 0) + d;
   986→      if (e._bpath >= BLEED_STEP) {
   987→        e._bpath = 0;
   988→        dropBlood(e._bx + dx * 0.5, e._by + dy * 0.5);
   989→      }
   990→      e._bx = e.x; e._by = e.y;
   991→    }
   992→    // старение меток
   993→    for (var k = splats.length - 1; k >= 0; k--) {
   994→      if (!splats[k].o || splats[k].o.removed) { splats.splice(k, 1); continue; }
   995→      if (now - splats[k].tmin > BLEED_MIN) {
   996→        try { WS.World.removeObject(splats[k].o); } catch (e2) { /* пусто */ }
   997→        splats.splice(k, 1);
   998→      }
   999→    }
  1000→  }

  1119→      }

  1121→      // пульс запаха: хищники чуют издалека
  1122→      b.pulse = (b.pulse || 0) + dt;

  1141→  function useDeerCall() {
  1142→    if (!running() || !uiFree()) return false;
  1143→    var p = P();
  1144→    if (p.sleeping) return false;
  1145→    if (hasItem('deer_call') <= 0) {
  1146→      toastOnce('nocall', t9('hunt.no_call', 'Нет манка (крафт: береста + нож)'), 'info');
  1147→      return false;
  1148→    }
  1149→    var nowS = Date.now();
  1150→    if (call && call.cd > nowS) {

  1152→      return false;

  1157→      call = { t: 0, deer: [], cd: nowS + CALL_CD * 500 };
  1158→      return true;
  1159→    }
  1160→    // собрать оленей в радиусе зова
  1161→    var deer = [];

  1164→      if (list[i] && list[i].kind === 'deer' && !list[i].dead) deer.push(list[i]);
  1165→    }
  1166→    call = { t: CALL_DUR, deer: deer, cd: nowS + CALL_CD * 1000 };
  1167→    fxText(t9('hunt.call_ok', 'Манк оленя!'), p.x, p.y - 1.1, { col: '#c8b46a', size: 13 });
  1168→    addXp('track', 1, 'зов оленя');
  1169→    if (!deer.length) {
  1170→      toastOnce('callnone', t9('hunt.call_none', 'Тишина. Оленей поблизости нет'), 'info');
  1171→    }
  1172→    return true;
  1173→  }

  1175→  /** Тик зова: олени бредут к игроку (не в панике). */
  1176→  function callTick(dt) {
  1177→    if (!call) return;
  1178→    call.t -= dt;
  1179→    if (call.t <= 0) { call = null; return; }
  1180→    var p = P();
  1181→    if (!p) { call = null; return; }
  1182→    for (var i = 0; i < call.deer.length; i++) {
  1183→      var e = call.deer[i];
  1184→      if (!e || e.dead) continue;
  1185→      var herd = e.herd;
  1186→      if (herd && herd.state === 'flee') continue;   // спугнуты — зов не действует
  1187→      // точка привязи: кольцо вокруг игрока, чуть дальше дневного чутья
  1188→      var a = Math.atan2(e.y - p.y, e.x - p.x) + (Math.random() - 0.5) * 0.6;
  1189→      var rr = CALL_KEEP_DIST + Math.random() * 1.5;
  1190→      e.wanderPt = {
  1191→        x: clamp(p.x + Math.cos(a) * rr, 4, 215),
  1192→        y: clamp(p.y + Math.sin(a) * rr * 0.85, 4, 215)
  1193→      };
  1194→      e.wanderWait = 0;
  1195→    }
  1196→  }
