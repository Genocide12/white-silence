    21→ *   • инструменты — по специализации (кузня, пилорама, дом).
    22→ *
    23→ * Связи: data/loot_tables.js (база), data/items_*.js (предметы),
    24→ * systems/looting.js (WS.Loot.fill читает WS.Data.loot[id]),
    25→ * content/quests.js (deserter_den, timo_cache — «поздние» таблицы),
    26→ * ai/hermit.js (hermit_herbs), data/loot_professions.js (трофеи).

    30→
    31→  // ====================================================================
    32→  //  ХЕЛПЕРЫ (формат тот же, что в loot_tables.js)
    33→  //  C(id, w, min, max)  — предмет с весом и количеством;
    34→  //  NOTHING(w)          — «пустой» слот (id: null);
    35→  //  T(rolls, items)     — заготовка новой таблицы.
    36→  // ====================================================================
    37→
    38→  function C(id, w, min, max) {
    39→    return { id: id, w: w, min: min || 1, max: max || min || 1 };
    40→  }
    41→  function NOTHING(w) { return { id: null, w: w, min: 0, max: 0 }; }
    42→  function T(rolls, items) { return { rolls: rolls, items: items }; }
    43→
    44→  // ====================================================================
    45→  //  WS.LOOTMERGE — модуль слияния лут-таблиц.
    46→  //
    47→  //  Задача: добавить предметы в УЖЕ существующую таблицу, не перезаписывая
    48→  //  её (лут-таблицы принадлежат loot_tables.js, quests.js, hermit.js —
    49→  //  править чужие файлы нельзя). Правила слияния:
    50→  //
    51→  //  1. Новый предмет просто дописывается в конец items.
    52→  //  2. Если предмет с таким id уже есть — вес не складываем, а берём
    53→  //     больший из двух (усиление, а не задвоение), диапазон количества
    54→  //     обновляем на новый.
    55→  //  3. После добавления пересчитываем вес «пусто» (id: null): шанс
    56→  //     пустого обыска ДОЛЖЕН остаться тем же, каким был до расширения.
    57→  //     Иначе каждая новая партия вещей превращала бы сундуки в золотые
    58→  //     жилы: в файле items_cloth.js 80+ вещей — без пересчёта изба стала
    59→  //     бы богаче губернского магазина.
    60→  //
    61→  //  Математика (для таблицы с суммарным весом вещей S и «пусто» N):
    62→  //     p_пусто_было = N / (N + S);
    63→  //  добавили новый вес A → новый масштаб (N + S + A), значит
    64→  //     N' = N * (N + S + A) / (N + S).
    65→  //  Старые предметы теряют долю ровно настолько, насколько входят новые,
    66→  //  «пусто» держит свою — раскладка честная.
    67→  //
    68→  //  Все функции безопасно звать только из boot или позже: на этапе
    69→  //  загрузки файла WS.Data ещё не существует (контракт ns.js), поэтому
    70→  //  сам модуль объявляем через WS.def, а работаем — в WS.boot.
    71→  // ====================================================================
    72→
    73→  /** Суммарный вес items по признаку «вещь / пусто». */
    74→  function weights(t) {
    75→    var res = { stuff: 0, nothing: 0 };
    76→    if (!t || !t.items) return res;
    77→    for (var i = 0; i < t.items.length; i++) {
    78→      var it = t.items[i];
    79→      if (!it) continue;
    80→      if (it.id === null || it.id === undefined) res.nothing += (it.w || 0);
    81→      else res.stuff += (it.w || 0);
    82→    }
    83→    return res;
    84→  }
    85→
    86→  /** Найти предмет по id в таблице (или null). */
    87→  function findItem(t, id) {
    88→    if (!t || !t.items) return null;
    89→    for (var i = 0; i < t.items.length; i++) {
    90→      if (t.items[i] && t.items[i].id === id) return t.items[i];
    91→    }
    92→    return null;
    93→  }
    94→
    95→  /** Пересчёт веса «пусто» под добавленную массу A (см. математику выше). */
    96→  function rescaleNothing(t, added) {
    97→    if (!t || added <= 0) return;
    98→    var w = weights(t);
    99→    if (w.nothing <= 0 || w.stuff <= 0) return;   // «пусто» нет — не трогаем
   100→    var scale = (w.nothing + w.stuff + added) / (w.nothing + w.stuff);
   101→    for (var i = 0; i < t.items.length; i++) {
   102→      var it = t.items[i];
   103→      if (it && (it.id === null || it.id === undefined)) it.w = (it.w || 0) * scale;
   104→    }
   105→  }
   106→
   107→  /** Нормализация записи предмета (min<=max, вес>=0). */
   108→  function normItem(o) {
   109→    var it = {
   110→      id: o.id,
   111→      w: (o.w === undefined || o.w === null) ? 1 : o.w,
   112→      min: (o.min === undefined || o.min === null) ? 1 : o.min,
   113→      max: (o.max === undefined || o.max === null) ? 1 : o.max
   114→    };
   115→    if (it.max < it.min) it.max = it.min;
   116→    if (it.w < 0) it.w = 0;
   117→    return it;
   118→  }
   119→
   120→  /**
   121→   * extend(tableId, newItems) -> {added, merged, ok} | null
   122→   *   tableId  — id существующей таблицы в WS.Data.loot;
   123→   *   newItems — массив записей C(...)/NOTHING(...) (хелперы выше);
   124→   *              записи с id:null игнорируются — «пусто» трогает только
   125→   *              пересчёт (см. rescaleNothing).
   126→   * Возвращает счётчик добавлений или null, если таблицы нет.
   127→   */
   128→  function extend(tableId, newItems) {
   129→    var loot = (WS.Data && WS.Data.loot) || null;
   130→    if (!loot) return null;
   131→    var t = loot[tableId];
   132→    if (!t || !t.items) return null;
   133→    var added = 0, merged = 0, mass = 0;
   134→    for (var i = 0; i < newItems.length; i++) {
   135→      var ni = newItems[i];
   136→      if (!ni || ni.id === null || ni.id === undefined) continue;
   137→      var ex = findItem(t, ni.id);
   138→      if (ex) {
   139→        // уже есть — усиливаем вес до большего из двух, обновляем диапазон
   140→        if ((ni.w || 0) > (ex.w || 0)) ex.w = ni.w;
   141→        if (ni.min !== undefined && ni.min !== null) ex.min = ni.min;
   142→        if (ni.max !== undefined && ni.max !== null) ex.max = ni.max;
   143→        if (ex.max < ex.min) ex.max = ex.min;
   144→        merged++;
   145→      } else {
   146→        t.items.push(normItem(ni));
   147→        mass += (ni.w || 0);          // масса только НОВЫХ записей
   148→        added++;
   149→      }
   150→    }
   151→    if (added > 0) rescaleNothing(t, mass);
   152→    return { added: added, merged: merged, ok: true };
   153→  }
   154→
   155→  /**
   156→   * add(tableId, rolls, items) -> true/false
   157→   *   Задекларировать НОВУЮ таблицу. Если таблица с таким id уже есть
   158→   *   (чужая — quests.js, hermit.js, loot_tables.js) — не трогаем,
   159→   *   вернём false: свои вещи в чужой карман не кладём.
   160→   */
   161→  function add(tableId, rolls, items) {
   162→    var loot = (WS.Data && WS.Data.loot) || null;
   163→    if (!loot || loot[tableId]) return false;
   164→    var t = { rolls: rolls || [1, 2], items: [] };
   165→    for (var i = 0; i < items.length; i++) {
   166→      if (!items[i]) continue;
   167→      t.items.push(normItem(items[i]));   // включая NOTHING-записи
   168→    }
   169→    loot[tableId] = t;
   170→    return true;
   171→  }
   172→
   173→  /** Есть ли таблица (для самопроверки и «поздних» расширений). */
   174→  function has(tableId) {
   175→    return !!(WS.Data && WS.Data.loot && WS.Data.loot[tableId]);
   176→  }
   177→
   178→  /** Список id всех таблиц (отладка). */
   179→  function ids() {
   180→    var out = [];
   181→    var loot = WS.Data && WS.Data.loot;
   182→    if (!loot) return out;
   183→    for (var k in loot) if (loot.hasOwnProperty(k)) out.push(k);
   184→    return out;
   185→  }

   189→    add: add,
   190→    has: has,

   193→  });

  1506→  /** Применить блок расширений, вернуть статистику. */
  1507→  function applyExt(block, label) {
  1508→    var applied = 0, missing = 0, items = 0;
  1509→    for (var tableId in block) {
  1510→      if (!block.hasOwnProperty(tableId)) continue;
  1511→      var res = extend(tableId, block[tableId]);
  1512→      if (res) { applied++; items += res.added; }
  1513→      else missing++;
  1514→    }
  1515→    if (missing) WS.warn('loot:tables2 (' + label + '): таблиц не найдено: ' + missing);
  1516→    return { applied: applied, items: items, missing: missing };
  1517→  }

  1519→  /** Применить все новые таблицы. */
  1520→  function applyNew() {
  1521→    var n = 0;
  1522→    for (var tableId in NEW) {
  1523→      if (!NEW.hasOwnProperty(tableId)) continue;
  1524→      if (add(tableId, NEW[tableId].rolls, NEW[tableId].items)) n++;
  1525→    }
  1526→    return n;
  1527→  }

  1529→  /**
  1530→   * Самопроверка (после первого кадра — все boot отработали, включая
  1531→   * recipes.js с fallback-предметами и loot_professions.js с трофеями):
  1532→   *   - каждая НАША таблица существует;
  1533→   *   - НОВЫЕ таблицы: каждый id существует, «пусто» присутствует;
  1534→   *   - РАСШИРЕНИЯ: проверяем только ДОБАВЛЕННЫЕ нами id (базовые
  1535→   *     записи loot_tables.js — не наша зона ответственности: там
  1536→   *     есть предметы «из i18n без спеки» — telegraph_form и пр.,
  1537→   *     движок их терпит осознанно).
  1538→   * Список проблем — в консоль (WS.warn), игра не валидится.
  1539→   */
  1540→  function selfTest() {
  1541→    var bad = [];
  1542→    var loot = (WS.Data && WS.Data.loot) || {};
  1543→    var items = (WS.Data && WS.Data.items) || {};
  1544→
  1545→    function checkId(where, id) {
  1546→      if (!items[id]) bad.push(where + ': нет предмета ' + id);
  1547→    }
  1548→
  1549→    // новые таблицы — полностью наши
  1550→    var k, i;
  1551→    for (k in NEW) {
  1552→      if (!NEW.hasOwnProperty(k)) continue;
  1553→      var t = loot[k];
  1554→      if (!t) { bad.push(k + ': таблица не объявлена'); continue; }
  1555→      var nothing = 0;
  1556→      for (i = 0; i < t.items.length; i++) {
  1557→        var it = t.items[i];
  1558→        if (!it) { bad.push(k + '[' + i + ']: пустая запись'); continue; }
  1559→        if (it.id === null || it.id === undefined) { nothing += (it.w || 0); continue; }
  1560→        checkId(k, it.id);
  1561→        if ((it.w || 0) < 0) bad.push(k + ': вес<0 у ' + it.id);
  1562→        if ((it.min || 0) > (it.max || 0)) bad.push(k + ': min>max у ' + it.id);
  1563→      }
  1564→      if (t.items.length && nothing <= 0) bad.push(k + ': нет «пусто»-слота');
  1565→    }
  1566→
  1567→    // расширения — только наши добавки
  1568→    for (k in EXT) {
  1569→      if (!EXT.hasOwnProperty(k)) continue;
  1570→      if (!loot[k]) { bad.push(k + ': расширение не нашло таблицу'); continue; }
  1571→      var list = EXT[k];
  1572→      for (i = 0; i < list.length; i++) {
  1573→        if (list[i] && list[i].id) checkId('ext:' + k, list[i].id);
  1574→      }
  1575→    }
  1576→    for (k in EXT_LATE) {
  1577→      if (!EXT_LATE.hasOwnProperty(k)) continue;
  1578→      if (!loot[k]) { bad.push('late:' + k + ': таблицы так и нет'); continue; }
  1579→      var list2 = EXT_LATE[k];
  1580→      for (i = 0; i < list2.length; i++) {
  1581→        if (list2[i] && list2[i].id) checkId('late:' + k, list2[i].id);
  1582→      }
  1583→    }
  1584→
  1585→    if (bad.length) {
  1586→      WS.warn('loot:tables2 self-test: ' + bad.length + ' проблем: ' + bad.join('; '));
  1587→    } else {
  1588→      WS.log('loot:tables2 self-test: OK');
  1589→    }
  1590→    return bad.length === 0;
  1591→  }
